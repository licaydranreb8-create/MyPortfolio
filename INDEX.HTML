//import java.util.Scanner;

public class DailyIncomeGoalMonitor {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        double[] income = new double[10];
        double totalIncome = 0;
        int daysMetTarget = 0;
        double target = 3000;

        // Input income for 10 days
        for (int i = 0; i < income.length; i++) {
            do {
                System.out.print("Enter income for Day " + (i + 1) + ": ");
                income[i] = input.nextDouble();

                if (income[i] < 0) {
                    System.out.println("Invalid input! Enter a nonnegative amount.");
                }
            } while (income[i] < 0);

            totalIncome += income[i];

            if (income[i] >= target) {
                daysMetTarget++;
            }
        }

        // Calculate average income
        double averageIncome = totalIncome / income.length;

        // Display results
        System.out.println("\n===== DAILY INCOME REPORT =====");
        System.out.printf("Total Income: PHP %.2f%n", totalIncome);
        System.out.printf("Average Daily Income: PHP %.2f%n", averageIncome);
        System.out.println("Days that met the target: " + daysMetTarget);

        // Display days that failed to reach the target
        System.out.println("Days that failed to reach PHP 3,000:");

        boolean failed = false;

        for (int i = 0; i < income.length; i++) {
            if (income[i] < target) {
                System.out.println("Day " + (i + 1) + " - PHP "
                        + String.format("%.2f", income[i]));
                failed = true;
            }
        }

        if (!failed) {
            System.out.println("None. All days reached the target!");
        }

        input.close();
    }
}
